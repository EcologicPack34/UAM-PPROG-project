function Add():
Input: Element element, Set set;
Output: Status;

Bool condition = FALSE;

foreach Element x in set:
	if isEqual(x, element) then:
		condition = TRUE
		break
	end
end

if condition == FALSE then:
	set[set.Length + 1] = element
	set.Length += 1
end

return OK
